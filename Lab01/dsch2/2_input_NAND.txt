module 2_input_NAND( A,B,Y);
 input A,B;
 output Y;
 nmos #(10) nmos(w2,vss,B); // 1.0u 0.12u
 nmos #(24) nmos(Y,w2,A); // 1.0u 0.12u

 pmos #(24) pmos(Y,vdd,B); // 2.0u 0.12u
endmodule

always
#1000 A=~A;
